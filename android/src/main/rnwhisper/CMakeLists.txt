cmake_minimum_required(VERSION 3.16)

# The rnwhisper core: whisper.cpp + ggml + cpp/rn-whisper.cpp, one shared
# library per CPU-feature variant. It has no React Native dependency, so it can
# be prebuilt once (scripts/build-android.sh -> android/src/main/jniLibs, shipped
# as a GitHub release asset) and reused across React Native versions. The
# JNI/JSI wrapper in ../CMakeLists.txt links against it, either as a sibling
# target (rnwhisperBuildFromSource=true) or as the prebuilt .so.
project(rnwhisper)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_ANDROID_STL_TYPE c++_shared)

include(${CMAKE_CURRENT_SOURCE_DIR}/../cmake/rnwhisper-build-options.cmake)
include(${CMAKE_CURRENT_SOURCE_DIR}/../../../../cmake/rnwhisper-sources.cmake)

if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(RNWHISPER_STANDALONE_BUILD ON)
    message(STATUS "rnwhisper: standalone build (prebuilt distribution)")
else ()
    set(RNWHISPER_STANDALONE_BUILD OFF)
    message(STATUS "rnwhisper: building as a subdirectory of the JNI wrapper")
endif ()

# Hexagon (Qualcomm NPU). The rnwhisper_*_hexagon variant is only built when
# the SDK is present: its host side needs the SDK's FastRPC headers and the
# stub that scripts/build-hexagon-htp.sh generates into htp/v73/ (the driver
# itself dlopens the device's libcdsprpc.so, see htp-drv.cpp). The DSP-side
# libggml-htp-*.so come from the same script and are shipped as app assets
# (see ../../../build.gradle).
set(HEXAGON_SDK_ROOT "" CACHE PATH "Hexagon SDK root directory")
set(RNWHISPER_HEXAGON_STUB_DIR "${RNWHISPER_GGML_HEXAGON_DIR}/htp/v73")
if (HEXAGON_SDK_ROOT AND IS_DIRECTORY "${HEXAGON_SDK_ROOT}/incs/stddef")
    set(RNWHISPER_HEXAGON_AVAILABLE ON)
    message(STATUS "rnwhisper: Hexagon SDK found at ${HEXAGON_SDK_ROOT}")
else ()
    set(RNWHISPER_HEXAGON_AVAILABLE OFF)
    message(STATUS "rnwhisper: Hexagon SDK not found, skipping the rnwhisper_*_hexagon variant")
endif ()

find_library(LOG_LIB log)

if (ANDROID_ABI STREQUAL "arm64-v8a")
    set(RNWHISPER_VAD_ARM_VARIANTS ON)
else ()
    set(RNWHISPER_VAD_ARM_VARIANTS OFF)
endif ()
rnwhisper_vad_configure_sources(${RNWHISPER_VAD_ARM_VARIANTS} RNWHISPER_VAD_ISA_SOURCES RNWHISPER_VAD_ISA_DEFINITIONS)

# --- sources shared by every variant ------------------------------------------
# A variant differs from the others only in its CPU flags (-march=...) and in
# the macros below. Three groups of sources can tell the difference and are
# compiled per variant:
#   - the ggml-cpu backend, whose SIMD kernels are selected by the CPU flags
#   - vad.cpp's vad-cpu backend, for the same reason
#   - sources that read one of these macros (e.g. backend registration)
# Everything else is compiled once per ABI, at the ABI's baseline, and linked
# into every variant (except on armeabi-v7a, see below). Upstream whisper.cpp
# scopes its CPU flags to ggml-cpu the same way.
#
# Every macro that only some variants define belongs in this list.
set(RNWHISPER_VARIANT_MACROS
    GGML_CPU_GENERIC
    GGML_USE_HEXAGON
)
string(REPLACE ";" "|" variant_macro_regex "${RNWHISPER_VARIANT_MACROS}")

set(RNWHISPER_VARIANT_SOURCES ${RNWHISPER_GGML_CPU_SOURCES} ${RNWHISPER_VAD_CPU_SOURCES})
# kernels_armv82.cpp (RNWHISPER_VAD_ISA_SOURCES) sets its own -march, which
# overrides the variant's, so it is the same object everywhere.
set(RNWHISPER_SHARED_SOURCES ${RNWHISPER_CORE_SOURCES} ${RNWHISPER_VAD_ISA_SOURCES})
list(REMOVE_ITEM RNWHISPER_SHARED_SOURCES ${RNWHISPER_VARIANT_SOURCES})
if (ANDROID_ABI STREQUAL "armeabi-v7a")
    # On 32-bit ARM the variant flag (-mfpu=neon-vfpv4) selects the FPU for all
    # floating-point code, not only for the SIMD kernels, and LLVM does not
    # inline between functions built for different FPUs. Sharing would change
    # the vfpv4 variant well beyond its kernels, so nothing is shared here.
    list(APPEND RNWHISPER_VARIANT_SOURCES ${RNWHISPER_SHARED_SOURCES})
    set(RNWHISPER_SHARED_SOURCES "")
endif ()
# The split below follows the contents of these sources, so redo it when one changes.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${RNWHISPER_SHARED_SOURCES})
foreach (source ${RNWHISPER_SHARED_SOURCES})
    file(STRINGS ${source} reads_variant_macro REGEX "${variant_macro_regex}" LIMIT_COUNT 1)
    if (reads_variant_macro)
        list(REMOVE_ITEM RNWHISPER_SHARED_SOURCES ${source})
        list(APPEND RNWHISPER_VARIANT_SOURCES ${source})
    endif ()
endforeach ()

# A shared source could still see one of the macros through a header, and would
# then silently be built without it for the variants that define it. Poisoning
# the names turns that into a compile error ("attempt to use a poisoned
# identifier"); the sources that hit it have to move to RNWHISPER_VARIANT_SOURCES.
string(REPLACE ";" " " poisoned_macros "${RNWHISPER_VARIANT_MACROS}")
set(RNWHISPER_SHARED_SOURCE_GUARD ${CMAKE_CURRENT_BINARY_DIR}/rnwhisper-shared-source-guard.h)
file(GENERATE OUTPUT ${RNWHISPER_SHARED_SOURCE_GUARD} CONTENT
"// Generated by android/src/main/rnwhisper/CMakeLists.txt, see RNWHISPER_VARIANT_MACROS there.
#pragma GCC poison ${poisoned_macros}
")

# What every object of every variant is compiled with.
set(RNWHISPER_COMPILE_DEFINITIONS
    _GNU_SOURCE  # Linux CPU affinity APIs used by ggml on Android
    # Only the public whisper/ggml/parakeet/vad/rnwhisper APIs are exported; the
    # *_SHARED/*_BUILD macros switch their API annotations to
    # visibility("default") while everything else stays hidden.
    GGML_SHARED GGML_BUILD
    WHISPER_SHARED WHISPER_BUILD
    PARAKEET_SHARED PARAKEET_BUILD
    VAD_SHARED VAD_BUILD
    RNWHISPER_SHARED RNWHISPER_BUILD
    ${RNWHISPER_VAD_DEFINITIONS}
    ${RNWHISPER_VAD_ISA_DEFINITIONS}
    ${RNWHISPER_VERSION_DEFINITIONS}
)
# ThinLTO rather than monolithic LTO: linking is parallel and incremental, and
# peak memory stays low.
set(RNWHISPER_COMPILE_OPTIONS
    -DGGML_USE_CPU -DGGML_USE_CPU_REPACK -pthread
    -O3 -DNDEBUG
    -fvisibility=hidden -fvisibility-inlines-hidden
    -ffunction-sections -fdata-sections
    -flto=thin
)
if (CMAKE_BUILD_TYPE AND CMAKE_BUILD_TYPE STREQUAL "Debug")
    list(APPEND RNWHISPER_COMPILE_OPTIONS -DRNWHISPER_ANDROID_ENABLE_LOGGING)
endif ()

set(RNWHISPER_SHARED_OBJECTS "")
if (RNWHISPER_SHARED_SOURCES)
    # EXCLUDE_FROM_ALL: only built for the variants that are enabled for this ABI.
    add_library(rnwhisper_shared_objects OBJECT EXCLUDE_FROM_ALL ${RNWHISPER_SHARED_SOURCES})
    target_include_directories(rnwhisper_shared_objects PRIVATE ${RNWHISPER_INCLUDE_DIRS})
    target_compile_definitions(rnwhisper_shared_objects PRIVATE ${RNWHISPER_COMPILE_DEFINITIONS})
    target_compile_options(rnwhisper_shared_objects PRIVATE
        ${RNWHISPER_COMPILE_OPTIONS} -include ${RNWHISPER_SHARED_SOURCE_GUARD})
    set_target_properties(rnwhisper_shared_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
    set(RNWHISPER_SHARED_OBJECTS $<TARGET_OBJECTS:rnwhisper_shared_objects>)
endif ()

# At every link, ThinLTO optimizes and generates code for each module again:
# about a third of a variant's build. lld can cache that per module, which the
# variants after the first then reuse for the shared objects, and rebuilds for
# whatever did not change. RNWHISPER_THINLTO_CACHE_DIR (CMake or environment)
# moves the cache out of the build tree, e.g. to persist it on CI.
#
# lld drops entries that went unused for a week, which a cache restored from an
# archive never looks like, so the size is capped as well: a few times what
# one build of every variant of every ABI needs.
set(RNWHISPER_THINLTO_CACHE_POLICY cache_size_bytes=1g)
if (NOT RNWHISPER_THINLTO_CACHE_DIR)
    set(RNWHISPER_THINLTO_CACHE_DIR "$ENV{RNWHISPER_THINLTO_CACHE_DIR}")
endif ()
if (NOT RNWHISPER_THINLTO_CACHE_DIR)
    set(RNWHISPER_THINLTO_CACHE_DIR "${CMAKE_BINARY_DIR}/thinlto-cache")
endif ()
message(STATUS "rnwhisper: ThinLTO cache at ${RNWHISPER_THINLTO_CACHE_DIR}")

function(build_rnwhisper_library target_name arch cpu_flags)
    rnwhisper_variant_enabled(${target_name} VARIANT_ENABLED)
    if (NOT VARIANT_ENABLED)
        message(STATUS "Skipping ${target_name} - not in RNWHISPER_ANDROID_VARIANTS")
        return()
    endif ()

    set(ENABLE_HEXAGON OFF)
    if (${target_name} MATCHES "_hexagon$")
        if (NOT RNWHISPER_HEXAGON_AVAILABLE)
            message(STATUS "Skipping ${target_name} - Hexagon SDK not available")
            return()
        endif ()
        set(ENABLE_HEXAGON ON)
    endif ()

    if (NOT ${arch} STREQUAL "generic")
        set(SOURCE_FILES_ARCH
            ${RNWHISPER_GGML_CPU_ARCH_DIR}/${arch}/quants.c
            ${RNWHISPER_GGML_CPU_ARCH_DIR}/${arch}/repack.cpp
        )
    endif ()

    # What this variant compiles for itself (see RNWHISPER_VARIANT_SOURCES). It
    # is a target of its own so that it never waits for another variant's link.
    set(variant_objects ${target_name}_objects)
    add_library(${variant_objects} OBJECT
        ${RNWHISPER_VARIANT_SOURCES}
        ${SOURCE_FILES_ARCH}
    )
    target_include_directories(${variant_objects} PRIVATE ${RNWHISPER_INCLUDE_DIRS})
    target_compile_definitions(${variant_objects} PRIVATE ${RNWHISPER_COMPILE_DEFINITIONS})
    target_compile_options(${variant_objects} PRIVATE ${RNWHISPER_COMPILE_OPTIONS} ${cpu_flags})
    set_target_properties(${variant_objects} PROPERTIES POSITION_INDEPENDENT_CODE ON)

    if (${arch} STREQUAL "generic")
        target_compile_options(${variant_objects} PRIVATE -DGGML_CPU_GENERIC)
    endif ()

    add_library(
        ${target_name}
        SHARED
        ${RNWHISPER_SHARED_OBJECTS}
        $<TARGET_OBJECTS:${variant_objects}>
    )
    set_target_properties(${target_name} PROPERTIES LINKER_LANGUAGE CXX)
    target_link_libraries(${target_name} PRIVATE ${LOG_LIB} android)

    if (ENABLE_HEXAGON)
        if (NOT EXISTS "${RNWHISPER_HEXAGON_STUB_DIR}/htp_iface_stub.c" OR NOT EXISTS "${RNWHISPER_HEXAGON_STUB_DIR}/htp_iface.h")
            message(FATAL_ERROR
                "Hexagon FastRPC stub is missing; run scripts/build-hexagon-htp.sh to generate ${RNWHISPER_HEXAGON_STUB_DIR}")
        endif ()

        target_sources(${variant_objects} PRIVATE
            ${RNWHISPER_GGML_HEXAGON_DIR}/ggml-hexagon.cpp
            ${RNWHISPER_GGML_HEXAGON_DIR}/htp-drv.cpp
            ${RNWHISPER_HEXAGON_STUB_DIR}/htp_iface_stub.c
        )
        target_include_directories(${variant_objects} PRIVATE
            ${HEXAGON_SDK_ROOT}/incs
            ${HEXAGON_SDK_ROOT}/incs/stddef
            ${HEXAGON_SDK_ROOT}/ipc/fastrpc/rpcmem/inc
            ${HEXAGON_SDK_ROOT}/utils/examples
            ${RNWHISPER_GGML_HEXAGON_DIR}
            ${RNWHISPER_GGML_HEXAGON_DIR}/htp
            ${RNWHISPER_HEXAGON_STUB_DIR}
        )
        target_link_libraries(${target_name} PRIVATE dl)
        target_compile_options(${variant_objects} PRIVATE -DGGML_USE_HEXAGON)
    endif ()

    target_link_options(${target_name} PRIVATE -Wl,--gc-sections)
    target_link_options(${target_name} PRIVATE -Wl,--exclude-libs,ALL)
    # ggml/whisper symbols keep upstream names. Bind this library's own
    # references at link time so a second ggml copy in the process (e.g.
    # llama.rn) can never be picked up via the dynamic linker's global lookup.
    target_link_options(${target_name} PRIVATE -Wl,-Bsymbolic)
    target_link_options(${target_name} PRIVATE -flto=thin
        -Wl,--thinlto-cache-dir=${RNWHISPER_THINLTO_CACHE_DIR}
        -Wl,--thinlto-cache-policy=${RNWHISPER_THINLTO_CACHE_POLICY})

    # Link the variants one after another. Each link already uses every core,
    # so running them side by side only multiplies peak memory, and a variant
    # can reuse the ThinLTO cache entries of the one linked before it.
    get_property(previous_variant GLOBAL PROPERTY RNWHISPER_PREVIOUS_VARIANT)
    if (previous_variant)
        add_dependencies(${target_name} ${previous_variant})
    endif ()
    set_property(GLOBAL PROPERTY RNWHISPER_PREVIOUS_VARIANT ${target_name})
endfunction()

rnwhisper_android_variants(RNWHISPER_VARIANTS)
foreach (entry IN LISTS RNWHISPER_VARIANTS)
    rnwhisper_split_variant("${entry}" core arch flags)
    build_rnwhisper_library(${core} ${arch} "${flags}")
endforeach ()
