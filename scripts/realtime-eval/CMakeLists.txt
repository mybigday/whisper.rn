# Host build of the realtime evaluation server (see README.md): the vendored
# whisper.cpp and ggml (CPU), plus the JSI layer's audioCtx rule.
cmake_minimum_required(VERSION 3.14)
project(realtime_eval C CXX)
set(CMAKE_CXX_STANDARD 17)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
set(W ${ROOT}/vendor/whisper.cpp)
set(GGML_NATIVE ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(${W}/ggml ggml)
add_library(whisper STATIC ${W}/src/whisper.cpp ${W}/src/aneforge/whisper-aneforge.cpp)
target_include_directories(whisper PUBLIC ${W}/include ${W}/src)
target_compile_definitions(whisper PRIVATE WHISPER_VERSION="realtime-eval")
target_link_libraries(whisper PUBLIC ggml)
add_executable(whisper-server whisper-server.cpp)
target_include_directories(whisper-server PRIVATE ${ROOT}/cpp/jsi)
target_link_libraries(whisper-server whisper)
add_library(parakeet STATIC ${W}/src/parakeet.cpp)
target_include_directories(parakeet PUBLIC ${W}/include ${W}/src)
target_compile_definitions(parakeet PRIVATE PARAKEET_VERSION="realtime-eval")
target_link_libraries(parakeet PUBLIC ggml)
add_executable(parakeet-server parakeet-server.cpp)
target_link_libraries(parakeet-server parakeet)

# Speech probabilities of a WAV with vad.cpp (same ggml), for a VAD on the host
set(VAD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(VAD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(VAD_BUILD_BENCH OFF CACHE BOOL "" FORCE)
add_subdirectory(${ROOT}/vendor/vad.cpp vad.cpp)
add_executable(vad-probs vad-probs.cpp)
target_link_libraries(vad-probs vad)
